if not transactions:
    return "没有要记的账目：transactions 为空。请向用户确认要记的内容和金额后重试。"
txns = []
for idx, t in enumerate(transactions, 1):
    if not isinstance(t, dict):
        return f"记账失败：第 {idx} 笔不是有效账目（需要含 type/amount/category 的对象），请核对后重试。"
    try:
        amount = float(t["amount"])
    except (KeyError, TypeError, ValueError):
        return f"记账失败：第 {idx} 笔金额无效（{t.get('amount')!r}），请向用户确认具体金额后重试。"
    txns.append(db.Transaction(
        type=str(t.get("type", "expense")),
        amount=amount,
        category=str(t.get("category", "其他")),
        note=str(t.get("note", "")),
        happened_at=str(t.get("happened_at") or "").strip(),
    ))
```

def _extract_reply(messages) -> str:
    if not messages:
        return ""
    last = messages[-1]
    if getattr(last, "type", "") == "ai" and getattr(last, "content", ""):
        return last.content
    return ""
```
